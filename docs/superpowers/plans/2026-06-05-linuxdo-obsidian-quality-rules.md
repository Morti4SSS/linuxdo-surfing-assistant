- [ ] **Step 2: Implement the minimal quality module**

Create `quality.py` with:

```python
from __future__ import annotations

from dataclasses import dataclass
import re


@dataclass(frozen=True)
class QualityIssue:
    code: str
    message: str
    page_name: str


BANNED_PHRASES = ("高相关", "累计权重")
OPAQUE_TERMS = ("风佬巨作", "zcf", "v5.0")
ELLIPSIS_PATTERN = re.compile(r"(?:\.\.\.|…)\s*(?:$|\n)")


def lint_human_markdown(text: str, *, page_name: str = "") -> list[QualityIssue]:
    issues: list[QualityIssue] = []
    for phrase in BANNED_PHRASES:
        if phrase in text:
            issues.append(QualityIssue("banned_phrase", f"包含旧噪音词：{phrase}", page_name))
    if ELLIPSIS_PATTERN.search(text):
        issues.append(QualityIssue("trailing_ellipsis", "包含省略号结尾的摘要", page_name))
    for term in OPAQUE_TERMS:
        if term in text:
            issues.append(QualityIssue("opaque_term", f"包含无上下文黑话：{term}", page_name))
    return issues
```

Extend `quality.py`:

```python
ALIASES = {
    "vibe-coding": ("vibecoding", "Vibecoding"),
    "vibecoding": ("vibecoding", "Vibecoding"),
    "ccswitch": ("cc-switch", "CC-Switch"),
    "cc-switch": ("cc-switch", "CC-Switch"),
    "vs code": ("vs-code", "VS Code"),
    "vscode": ("vs-code", "VS Code"),
}

BROAD_COLLECTIONS = {"公益站", "中转站", "API 中转"}
SERVICES = {"new api", "oneapi", "openrouter", "cliproxyapi", "cc-switch", "anyrouter"}
WORKFLOWS = {"superpowers", "trellis", "codestable", "ccw", "ccg", "bmad"}


def normalize_resource_name(name: str) -> tuple[str, str]:
    display = " ".join(str(name).strip().split())
    key = display.lower().replace("_", "-")
    key = " ".join(key.split())
    return ALIASES.get(key, (key, display))


def classify_knowledge_object(name: str) -> str:
    key, display = normalize_resource_name(name)
    if display in BROAD_COLLECTIONS:
        return "collection"
    if key in SERVICES:
        return "service"
    if key in WORKFLOWS:
        return "workflow"
    return "resource"
```
